cmake_minimum_required(VERSION 3.16)
project(intramural_sim)
find_package(ament_cmake REQUIRED)
find_package(ignition-gazebo6 REQUIRED)
find_package(ignition-transport11 REQUIRED)
add_library(intramural_drive SHARED drive.cpp)
target_compile_features(intramural_drive PRIVATE cxx_std_17)
target_compile_options(intramural_drive PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(intramural_drive ignition-gazebo6::ignition-gazebo6 ignition-transport11::core)
install(TARGETS intramural_drive DESTINATION lib)
install(FILES LICENSE DESTINATION share/${PROJECT_NAME})
ament_package()
