cmake_minimum_required(VERSION 3.16)
project(intramural_sim)
find_package(ament_cmake REQUIRED)
find_package(ignition-gazebo6 REQUIRED)
find_package(ignition-transport11 REQUIRED)
find_package(rclcpp REQUIRED)
find_package(ros_gz_bridge REQUIRED)
add_executable(sensor_bridge bridge.cpp)
ament_target_dependencies(sensor_bridge rclcpp ros_gz_bridge)
target_compile_features(sensor_bridge PRIVATE cxx_std_17)
install(TARGETS sensor_bridge DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_bridge test_bridge.cpp)
  ament_target_dependencies(test_bridge rclcpp ros_gz_bridge)
  add_test(NAME bridge_lifetime COMMAND test_bridge)
  set_tests_properties(bridge_lifetime PROPERTIES ENVIRONMENT "ROS_DOMAIN_ID=94;ROS_LOCALHOST_ONLY=1")
endif()
add_library(intramural_drive SHARED drive.cpp)
target_compile_features(intramural_drive PRIVATE cxx_std_17)
target_compile_options(intramural_drive PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(intramural_drive ignition-gazebo6::ignition-gazebo6 ignition-transport11::core)
install(TARGETS intramural_drive DESTINATION lib)
install(FILES LICENSE DESTINATION share/${PROJECT_NAME})
ament_package()
