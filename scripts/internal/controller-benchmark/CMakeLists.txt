cmake_minimum_required(VERSION 3.16)
project(intramural_controller_benchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(nav2_mppi_controller REQUIRED)
find_package(nav2_core REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(nlohmann_json REQUIRED)
add_executable(controller-worker main.cpp)
ament_target_dependencies(controller-worker nav2_mppi_controller nav2_core nav2_costmap_2d pluginlib tf2_geometry_msgs)
target_link_libraries(controller-worker nlohmann_json::nlohmann_json)
target_compile_options(controller-worker PRIVATE -O3 -Wall -Wextra)
